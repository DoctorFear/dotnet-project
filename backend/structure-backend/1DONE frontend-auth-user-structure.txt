frontend/src/
│
├── types/
│   └── user.ts
│       ├── Auth DTOs:
│       │   ├── LoginRequestDto               (Username, Password)

│       │   ├── RegisterRequestDto            (FullName, Username, Email, PhoneNumber, Password, ConfirmPassword)
│       │   ├── RegisterResponseDto           (UserId, Username, Email, FullName, Role, FPoints, CreatedAt)
│       │   ├── ForgotPasswordRequestDto      (Username, Email)
│       │   └── ResetPasswordRequestDto       (Token, NewPassword, ConfirmPassword)
│       ├── Profile DTOs:
│       │   ├── AddressDto                    (Province, Ward, StreetAddress, IsDefault)
│       │   ├── UserProfileResponseDto        (Id, Username, Email, FullName, PhoneNumber, Role, HasPassword, Addresses[])

│       │   └── ChangePasswordRequestDto      (CurrentPassword, NewPassword, ConfirmNewPassword)

│           ├── UserInfo, UserRole ('Member' | 'Staff' | 'Shipper' | 'Admin')
│           └── ApiResponse<T> (success, message, data)

├── services/
│   ├── apiClient.ts
│   │   ├── Tự động gắn Bearer Token vào header Authorization cho mọi request
│   │   ├── Silent Refresh Token: Bắt mã lỗi 401 tự gọi /Auth/refresh-token đổi AccessToken mới
│   │   ├── Chống Race Condition: Dùng Promise locking (refreshPromise) khi nhiều request 401 cùng lúc
│   │   └── Tự động Logout & chuyển hướng về /login khi Refresh Token hết hạn hoặc không hợp lệ
│   │
│   ├── authService.ts
│   │   ├── login(payload)                    -> POST /api/Auth/login
│   │   ├── register(payload)                 -> POST /api/Auth/register
│   │   ├── logout()                          -> POST /api/Auth/logout
│   │   ├── refreshToken(token)               -> POST /api/Auth/refresh-token
│   │   ├── forgotPassword(payload)           -> POST /api/Auth/forgot-password
│   │   ├── verifyResetToken(token)           -> GET  /api/Auth/verify-reset-token?token=...
│   │   └── resetPassword(payload)            -> POST /api/Auth/reset-password
│   │
│   └── userService.ts
│       ├── getProfile()                      -> GET  /api/User/profile
│       ├── updateProfile(payload)            -> PUT  /api/User/profile
│       └── changePassword(payload)           -> POST /api/User/change-password
│
├── context/
│   └── UserContext.tsx
│       ├── Quản lý Global State: currentUser, accessToken, refreshToken, isLoggedIn
│       ├── Đồng bộ tức thì với localStorage ('currentUser', 'accessToken', 'refreshToken')
│       └── Cung cấp các action:
│           ├── loginSuccess(authData)        -> Lưu state & localStorage khi đăng nhập thành công
│           ├── logoutClient()                -> Xóa sạch token & state khi đăng xuất
│           └── updateUser(partialInfo)       -> Cập nhật FullName, Email trên toàn hệ thống mà không cần reload
│
├── hooks/
│   ├── useToastNotifications.ts

│   │
│   └── useAuthAction.ts
│       ├── Xử lý các luồng nghiệp vụ Async: handleLogin, handleRegister, handleLogout,
│       │   handleForgotPassword, handleResetPassword
│       └── Tự động quản lý cờ isLoading, bắn Toast thông báo và điều hướng (navigate) theo Role:
│           ├── Admin   -> /admin/settings
│           ├── Staff   -> /staff/books
│           ├── Shipper -> /shipper/orders
│           └── Member  -> Trang trước đó hoặc Trang chủ (/)
